## Interfaces
config system interface
    edit "port1"
        set vdom "root"
        set ip 200.6.93.6 255.255.255.252
        set allowaccess ping http
        set type physical
        set snmp-index 1
    next
    edit "port2"
        set vdom "root"
        set ip 10.6.93.129 255.255.255.240
        set allowaccess ping
        set type physical
        set device-identification enable
        set lldp-transmission enable
        set role lan
        set snmp-index 2
    next
    edit "VPN-REMOTE"
        set vdom "root"
        set ip 169.254.1.1 255.255.255.255
        set allowaccess fabric
        set type tunnel
        set remote-ip 169.254.1.1 255.255.255.255
        set snmp-index 15
        set interface "port1"
    next
end

## VIP HTTPS
config firewall vip
    edit "VIP-HTTPS"
        set extip 200.6.93.6
        set mappedip "10.6.93.130"
        set extintf "port1"
        set portforward enable
        set extport 443
        set mappedport 443
    next
end

## Firewall policies
config firewall policy
    edit 1
        set name "HTTPS-SIN-VPN"
        set srcintf "port1"
        set dstintf "port2"
        set action accept
        set srcaddr "all"
        set dstaddr "VIP-HTTPS"
        set schedule "always"
        set service "HTTPS"
    next
    edit 2
        set name "vpn_VPN-REMOTE_remote_0"
        set srcintf "VPN-REMOTE"
        set dstintf "port2"
        set action accept
        set srcaddr "VPN-REMOTE_range"
        set dstaddr "port2 address"
        set schedule "always"
        set service "SSH"
        set nat enable
    next
end
